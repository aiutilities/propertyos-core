
--- backend/Dockerfile
     1	FROM node:22-alpine AS deps
     2	WORKDIR /app
     3	COPY package*.json ./
     4	COPY core-contracts ./core-contracts
     5	RUN npm ci
     6
     7	FROM node:22-alpine AS build
     8	WORKDIR /app
     9	COPY --from=deps /app/node_modules ./node_modules
    10	COPY --from=deps /app/core-contracts ./core-contracts
    11	COPY package*.json ./
    12	COPY tsconfig.json ./
    13	COPY src ./src
    14	RUN npm run build
    15	RUN npm prune --omit=dev
    16
    17	FROM node:22-alpine AS runtime
    18	WORKDIR /app
    19
    20	ENV NODE_ENV=production
    21
    22	RUN apk add --no-cache dumb-init
    23
    24	COPY --from=build --chown=node:node /app/package*.json ./
    25	COPY --from=build --chown=node:node /app/node_modules ./node_modules
    26	COPY --from=build --chown=node:node /app/core-contracts ./core-contracts
    27	COPY --from=build --chown=node:node /app/dist ./dist
    28	COPY --chown=node:node src/database/migrations ./src/database/migrations
    29
    30	RUN mkdir -p /app/uploads /app/plugins/.installed \
    31	  && chown -R node:node /app
    32
    33	USER node
    34
    35	EXPOSE 3000
    36
    37	HEALTHCHECK --interval=30s --timeout=5s --start-period=30s --retries=5 \
    38	  CMD node -e "fetch('http://127.0.0.1:3000/api/v1/health/ready').then(async r=>{const b=await r.json();process.exit(r.ok&&b.status==='ok'?0:1)}).catch(()=>process.exit(1))"
    39
    40	ENTRYPOINT ["dumb-init", "--"]
    41	CMD ["node", "dist/main.js"]
